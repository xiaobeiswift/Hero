func _run() -> void:
	if not _guard():
		push_error("ERROR: save folio native guard rejected before game load")
		quit(2); return
	guard_passed = true
	create_timer(180.0).timeout.connect(func():
		if not finished:
			_check(false,"Save folio capture exceeded 180-second watchdog")
			_finish())
	if not _check(DisplayServer.get_name() != "headless","Actual native display required"): _finish(); return
	if not _check(DirAccess.make_dir_absolute(frames_dir) == OK,"Create exclusive native frame directory"): _finish(); return
	if not await _boot(): _finish(); return
	if not _check(String(ProjectSettings.get_setting("application/config/version", "")) == "0.0.33", "Actual Web33 runtime version"): _finish(); return
	await _click("踏入江湖")
	if not _check(app.current_screen == "explore" and not app.active_modal,"Real title input starts production journey"): _finish(); return
	await _save_list_and_overwrite()
	if not failures.is_empty(): _finish(); return
	await _occupied_failure()
	if not failures.is_empty(): _finish(); return
	await _load_states()
	if not failures.is_empty(): _finish(); return
	await _failed_title_load()
	_check(captures.size() == FRAME_LIMIT,"Exactly twelve new native PNGs")
	_finish()

func _page():
	return app.overlay.get_node_or_null("SaveFolioPage")

func _click(text: String) -> void:
	var matches: Array[Button] = []
	for node in app.overlay.find_children("*","Button",true,false):
		if node.text == text and node.is_visible_in_tree() and not node.disabled: matches.append(node)
	if not _check(matches.size() == 1,"One real visible button: "+text): return
	await _pointer(matches[0],text)

func _kind(expected: String) -> bool:
	var page = _page()
	return _check(page != null and app.active_modal and page.data.get("kind", "") == expected and page.frame.is_visible_in_tree(),"Actual visible folio kind: "+expected)

func _native_focus(text: String) -> bool:
	var page = _page()
	if not _check(page != null,"Native focus begins on save folio"): return false
	# Always traverse at least once: pointer focus alone need not paint a ring.
	for attempt: int in range(page.buttons.size() + 1):
		await _key(KEY_TAB)
		var owner = root.gui_get_focus_owner()
		if owner is Button and owner.text == text:
			return _check(owner in page.buttons and owner.is_visible_in_tree(),"Native Tab reached visible "+text)
	return _check(false,"Native Tab could not reach "+text)

func _retain_then_replace(path: String, leaf: String, evidence_name: String, replacement: PackedByteArray, reason: String) -> bool:
	if not _owned_file(path,leaf,true): return false
	var original = FileAccess.get_file_as_bytes(path)
	var retained = owned_root.path_join("results/"+evidence_name+".json")
	if not _check(not original.is_empty() and evidence_name == evidence_name.get_file() and _no_links(retained) and not FileAccess.file_exists(retained) and not DirAccess.dir_exists_absolute(retained),"Fresh external original-byte retention destination"): return false
	var file = FileAccess.open(retained,FileAccess.WRITE)
	if not _check(file != null,"Open original-byte retention evidence"): return false
	file.store_buffer(original); file.flush(); var retain_error = file.get_error(); file.close()
	if not _check(retain_error == OK and FileAccess.get_file_as_bytes(retained) == original,"Complete original bytes retained before fixture corruption"): return false
	if not _owned_file(path,leaf,true) or not _check(FileAccess.get_file_as_bytes(path) == original,"Recheck exact unchanged fixture immediately before replacement"): return false
	file = FileAccess.open(path,FileAccess.WRITE)
	if not _check(file != null,"Open exact owned fixture for prepared replacement"): return false
	file.store_buffer(replacement); file.flush(); var replace_error = file.get_error(); file.close()
	if not _check(replace_error == OK and FileAccess.get_file_as_bytes(path) == replacement,"Prepared replacement bytes verified"): return false
	fixtures.append({"label":reason,"kind":"explicit_prepared_file_replacement_not_gameplay","path":ProjectSettings.globalize_path(path),"retained_original":retained,"retained_original_sha256":FileAccess.get_sha256(retained),"retained_original_base64":Marshalls.raw_to_base64(original),"application_files":_file_records()})
	return true

	if not guard_passed or not _no_links(report_path): return false
	if not failures.is_empty(): status = "fail"
	var report = {"suite":"save_folio_native_capture","status":status,"historical_evidence_recreated":false,"checks":checks,"failures":failures,"binding":binding,"display_server":DisplayServer.get_name(),"engine_version":Engine.get_version_info(),"captures":captures,"fixtures":fixtures,"inputs":inputs,"preservation_boundaries":boundaries,"checkpoints":checkpoints,"application_files":_file_records(),"limits":["Twelve new PNGs cover six approved states at two actual physical sizes with logical1280x800","Explicit fixture preparation and filesystem faults are not earned gameplay","Native scripted viewport input is not manual OS/browser input or Windows certification","Exact unchanged bytes cannot prove absence of same-content writes","Independent pixel inspection is required; control rectangles alone do not establish readability","No new assets, video, baseline replay, package, source editing or publication","Outer launcher verifies immutable original source/runtime/engine/driver/helper/binding after exit"]}
	var file = FileAccess.open(report_path,FileAccess.WRITE)
	if file == null: return false
	file.store_string(JSON.stringify(report,"\t")); file.flush()
	var error = file.get_error(); file.close()
	return error == OK

func _finish() -> void:
	if finished: return
	finished = true
	if collision_owned: _remove_collision()
	if is_instance_valid(app): app._stop_audio()
	var report_ok = _write_native_report("pass" if failures.is_empty() else "fail")
	if is_instance_valid(app): app.free()
	if not report_ok: quit(2); return
	if failures.is_empty(): print("PASS: NEW save folio native capture, %d PNGs / %d checks" % [captures.size(),checks])
	else: push_error("ERROR: save folio native capture failed, %d failures" % failures.size())
	quit(0 if failures.is_empty() else 1)

func _guard() -> bool:
	if not super._guard(): return false
	if OS.get_name() != "Linux": return false
	if OS.get_environment("HERO_FOLIO_QA_TOKEN").length() != 64: return false
	if owned_user != owned_root.path_join(USER_SUFFIX): return false
	if report_path != owned_root.path_join("results/test-result.json"): return false
	for entry: Array in [["HOME","home"],["XDG_DATA_HOME","data"],["XDG_CONFIG_HOME","config"],["XDG_CACHE_HOME","cache"]]:
		if OS.get_environment(entry[0]) != owned_root.path_join(entry[1]): return false
	frames_dir = owned_root.path_join("results/native-frames")
	if not _no_links(frames_dir) or FileAccess.file_exists(frames_dir) or DirAccess.dir_exists_absolute(frames_dir): return false
	var binding_path = owned_root.path_join("results/native-binding.json")
	if OS.get_environment("HERO_FOLIO_QA_NATIVE_BINDING") != binding_path or not _no_links(binding_path) or not FileAccess.file_exists(binding_path): return false
	if FileAccess.get_sha256(binding_path) != OS.get_environment("HERO_FOLIO_QA_NATIVE_BINDING_SHA256"): return false
	var parsed = JSON.parse_string(FileAccess.get_file_as_string(binding_path))
	if not parsed is Dictionary: return false
	binding = parsed
	if binding.get("suite") != "save_folio_native_capture": return false
	if binding.get("source") != ProjectSettings.globalize_path("res://").trim_suffix("/").simplify_path(): return false
	if binding.get("driver_sha256") != FileAccess.get_sha256(get_script().resource_path): return false
	if binding.get("helper_sha256") != FileAccess.get_sha256(HELPER_PATH): return false
	if binding.get("engine_sha256") != FileAccess.get_sha256(OS.get_executable_path()): return false
	for field: String in ["runtime_sha256"]:
		if not binding.get(field,"") is String or binding[field].length() != 64: return false
	for field: String in ["source_commit","source_tree"]:
		if not binding.get(field,"") is String or binding[field].length() != 40: return false
	if not binding.get("runtime_entries",0) is int and not binding.get("runtime_entries",0) is float: return false
	if int(binding.runtime_entries) <= 0: return false
	# The outer launcher independently pins the final runtime manifest and exact
	# WIP source tree. This driver never inherits the old recovery runtime hash.
	return _tree_unlinked(owned_user)


func _file_records(path: String = "", relative: String = "") -> Dictionary:
	var actual = owned_user if path.is_empty() else path
	var result: Dictionary = {}
	if not _check(_tree_unlinked(actual),"Application tree remains nonsymlink"): return result
	var directory = DirAccess.open(actual)
	if directory == null: return result
	directory.include_hidden = true; directory.include_navigational = false
	for leaf: String in directory.get_files():
		var filename = actual.path_join(leaf)
		var file = FileAccess.open(filename,FileAccess.READ)
		result[relative.path_join(leaf)] = {"bytes":file.get_length() if file != null else -1,"sha256":FileAccess.get_sha256(filename)}
		if file != null: file.close()
	for leaf: String in directory.get_directories():
		if relative.is_empty() and leaf in ["shader_cache","logs"]: continue
		result.merge(_file_records(actual.path_join(leaf),relative.path_join(leaf)))
	return result

func _rect(rect: Rect2) -> Array:
	return [rect.position.x,rect.position.y,rect.size.x,rect.size.y]

func _controls(node: Node) -> Array[Dictionary]:
	var rows: Array[Dictionary] = []
	if node is Control and node.is_visible_in_tree():
		var row: Dictionary = {"path":str(node.get_path()),"logical_rect":_rect(node.get_global_rect()),"pixel_rect":_rect(root.get_final_transform()*node.get_global_rect()),"focus":node.has_focus()}
		if node is Label or node is RichTextLabel or node is Button:
			row.text = node.text; row.font_size = node.get_theme_font_size("normal_font_size" if node is RichTextLabel else "font_size")
		if node is RichTextLabel: row.content_height = node.get_content_height(); row.viewport_height = node.size.y; row.scroll_offset = node.get_v_scroll_bar().value
		if node is Label: row.visible_line_count = node.get_visible_line_count(); row.line_count = node.get_line_count()
		if node is Button: row.disabled = node.disabled; row.pressed = node.button_pressed
		if node is ProgressBar: row.value = node.value; row.max_value = node.max_value
		if node is TextureRect and node.texture is AtlasTexture: row.atlas_region = _rect(node.texture.region); row.stretch_mode = node.stretch_mode
		rows.append(row)
	for child in node.get_children(): rows.append_array(_controls(child))
	return rows
